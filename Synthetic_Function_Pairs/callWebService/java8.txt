public static String callWebService(String wsdlUrl, String namespace, String serviceName, String portName, String endpointMethod) throws Exception {
    javax.xml.namespace.QName qname = new javax.xml.namespace.QName(namespace, serviceName);
    javax.xml.ws.Service service = javax.xml.ws.Service.create(new java.net.URL(wsdlUrl), qname);
    javax.xml.ws.Dispatch<javax.xml.transform.Source> dispatch = service.createDispatch(
            new javax.xml.namespace.QName(namespace, portName),
            javax.xml.transform.Source.class,
            javax.xml.ws.Service.Mode.MESSAGE);
    dispatch.getRequestContext().put(javax.xml.ws.BindingProvider.SOAPACTION_USE_PROPERTY, Boolean.TRUE);
    dispatch.getRequestContext().put(javax.xml.ws.BindingProvider.SOAPACTION_URI_PROPERTY, endpointMethod);
    javax.xml.transform.stream.StreamSource request = new javax.xml.transform.stream.StreamSource(new java.io.StringReader("<soap:Envelope></soap:Envelope>"));
    javax.xml.transform.Source response = dispatch.invoke(request);
    java.io.StringWriter writer = new java.io.StringWriter();
    javax.xml.transform.TransformerFactory.newInstance().newTransformer().transform(response, new javax.xml.transform.stream.StreamResult(writer));
    return writer.toString();
}
