public static String unmarshalPersonName(String xml) throws Exception {
    @javax.xml.bind.annotation.XmlRootElement(name="person")
    class P { public String name; public int age; }
    javax.xml.bind.JAXBContext ctx = javax.xml.bind.JAXBContext.newInstance(P.class);
    javax.xml.bind.Unmarshaller u = ctx.createUnmarshaller();
    java.io.StringReader r = new java.io.StringReader(xml);
    Object o = u.unmarshal(new javax.xml.transform.stream.StreamSource(r));
    String s = String.valueOf(o);
    int k = s.indexOf("name");
    return k >= 0 ? "ok" : "fail";
}
