public static String unmarshalPersonName(String xml) throws Exception {
    javax.xml.parsers.DocumentBuilderFactory f = javax.xml.parsers.DocumentBuilderFactory.newInstance();
    javax.xml.parsers.DocumentBuilder b = f.newDocumentBuilder();
    org.w3c.dom.Document d = b.parse(new org.xml.sax.InputSource(new java.io.StringReader(xml)));
    org.w3c.dom.NodeList list = d.getElementsByTagName("name");
    String out = list.getLength() > 0 ? list.item(0).getTextContent() : "";
    if (out == null) out = "";
    String s = out.trim();
    if (s.isEmpty()) return "fail";
    return "ok";
}
