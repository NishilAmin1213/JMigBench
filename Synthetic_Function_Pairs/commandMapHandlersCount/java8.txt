public static int commandMapHandlersCount() {
    javax.activation.MailcapCommandMap m = new javax.activation.MailcapCommandMap();
    m.addMailcap("text/plain;; x-java-content-handler=h");
    javax.activation.CommandInfo[] infos = m.getPreferredCommands("text/plain");
    int c = 0;
    for (javax.activation.CommandInfo i : infos) {
        if (i != null) c++;
    }
    return c;
}
