public static String marshalPersonXml(String n, int a) {
    StringBuilder sb = new StringBuilder();
    sb.append("<person>");
    sb.append("<name>").append(n == null ? "" : n).append("</name>");
    sb.append("<age>").append(a).append("</age>");
    sb.append("</person>");
    String s = sb.toString();
    if (s.length() == 0) {
        return "";
    } else {
        return s;
    }
}
