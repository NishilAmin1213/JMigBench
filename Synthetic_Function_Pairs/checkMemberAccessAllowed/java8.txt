public static boolean checkMemberAccessAllowed(Class<?> target) {
    SecurityManager sm = System.getSecurityManager();
    if (sm != null) {
        try {
            sm.checkMemberAccess(target, java.lang.reflect.Member.PUBLIC); // deprecated SecurityManager path
            return true;
        } catch (SecurityException se) {
            return false;
        }
    }
    return true;
}
