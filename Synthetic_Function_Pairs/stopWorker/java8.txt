public static void stopWorker() throws InterruptedException {
    Thread worker = new Thread(() -> {
        while (true) {
            try {
                Thread.sleep(200);
            } catch (InterruptedException e) {
                break;
            }
        }
    });
    worker.start();
    Thread.sleep(500);
    worker.stop();
    worker.join();
    System.out.println("Worker stopped.");
}
