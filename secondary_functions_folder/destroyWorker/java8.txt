public static void destroyWorker() throws InterruptedException {
    Thread worker = new Thread(() -> {
        while (true) {
            try {
                Thread.sleep(200);
            } catch (InterruptedException e) {
                break;
            }
        }
    });
    worker.start();
    Thread.sleep(500);
    worker.destroy();
    worker.join(200);
    System.out.println("Worker destroyed.");
}
