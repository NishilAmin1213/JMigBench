public static String createTempAndScheduleCleanup() throws java.io.IOException {
    java.io.File tmp = java.io.File.createTempFile("migrate_", ".tmp");
    java.io.OutputStream out = new java.io.FileOutputStream(tmp);
    byte[] data = "x".getBytes(java.nio.charset.StandardCharsets.UTF_8);
    out.write(data);
    out.flush();
    out.close();
    tmp.deleteOnExit();
    System.runFinalizersOnExit(true);
    return tmp.getAbsolutePath();
}
