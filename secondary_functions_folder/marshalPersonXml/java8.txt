public static String marshalPersonXml(String n, int a) throws Exception {
    @javax.xml.bind.annotation.XmlRootElement(name="person")
    class P { public String name; public int age; }
    P p = new P();
    p.name = n;
    p.age = a;
    javax.xml.bind.JAXBContext ctx = javax.xml.bind.JAXBContext.newInstance(P.class);
    javax.xml.bind.Marshaller m = ctx.createMarshaller();
    java.io.StringWriter w = new java.io.StringWriter();
    m.marshal(p, w);
    String s = w.toString();
    return s.length() > 0 ? s : "";
}
