public static void copyFile(String sourcePath, String targetPath) throws java.io.IOException {
    java.io.InputStream in = java.nio.file.Files.newInputStream(java.nio.file.Paths.get(sourcePath));
    java.io.OutputStream out = java.nio.file.Files.newOutputStream(java.nio.file.Paths.get(targetPath));
    byte[] buffer = new byte[8192];
    int read;
    while ((read = in.read(buffer)) != -1) {
        out.write(buffer, 0, read);
    }
    out.flush();
    in.close();
    out.close();
}
